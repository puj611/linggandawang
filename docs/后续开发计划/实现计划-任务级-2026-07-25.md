```typescript
export function extractCompletedTopics(
  requirements: Requirement[],
  scanResult?: FullScanResult | null,
): string[] {
  const topics = new Set<string>();
  
  // 1. 从已完成子任务提取（保持现有逻辑）
  for (const req of requirements) {
    for (const st of req.subtasks) {
      if (st.status !== 'done' && st.status !== 'skipped') continue;
      const titleWords = st.title
        .replace(/[（）()【】\[\]]/g, ' ')
        .split(/[\s,，、/]+/)
        .map((w) => w.trim().toLowerCase())
        .filter((w) => w.length >= 2 && !STOP_WORDS.has(w));
      for (const w of titleWords) topics.add(w);
      topics.add(st.type);
    }
  }
  
  // 2. 从项目扫描 TODO 提取已完成的主题
  if (scanResult?.todos) {
    for (const todo of scanResult.todos) {
      // status 为 done/removed 的视为已完成
      if (todo.status !== 'done' && todo.status !== 'removed') continue;
      // 从 TODO 文本提取关键词
      const text = todo.text
        .replace(/^DONE:\s*/i, '')
        .replace(/[（）()【】\[\]]/g, ' ')
        .split(/[\s,，、/]+/)
        .map((w) => w.trim().toLowerCase())
        .filter((w) => w.length >= 2 && !STOP_WORDS.has(w));
      for (const w of text) topics.add(w);
    }
  }
  
  return Array.from(topics);
}
```
